-/-/-/-/-/-/-/-/-/-/-/-/-/-/-/-/-/-/-/-/-/-/- VARIABLE -/-/-/-/-/-/-/-/-/-/-/-/-/-/-/-/-/-/-/-/-/-/-

--> variable is defined using two dashes (--)
--> var() fn is used to take value of 
--> to declare global variable (:root) selectors is used

syntax:
    global variable
        :root 
        {
            --b:blue
        }
        use of variable 
        p{
            color: var(--b)
        }
    local variable
    p{
        --b:blue
        color:var(--b)
    }
